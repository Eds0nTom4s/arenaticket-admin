1. VISÃO GERAL DO MÓDULO
================================================================================

OBJETIVO:
  Criar interface administrativa completa para gestão de usuários do sistema,
  permitindo criar, editar, listar, ativar/desativar usuários.

PERFIS DE USUÁRIO (Roles):
  - ADMIN: Administrador do sistema (acesso total)
  - VENDEDOR: Vendedor de bilhetes presenciais
  - PORTEIRO: Validador de bilhetes na entrada
  - MONITORING: Usuário técnico (métricas/logs)

FUNCIONALIDADES PRINCIPAIS:
  ✓ Listagem de usuários (tabela paginada com filtros)
  ✓ Busca por nome/telefone
  ✓ Filtro por role e status (ativo/inativo)
  ✓ Criação de novo usuário
  ✓ Edição de usuário existente
  ✓ Ativação/Desativação de usuário
  ✓ Visualização de detalhes do usuário
  ✓ Indicadores visuais de status

ACESSO:
  - Apenas usuários com role ADMIN podem acessar este módulo
  - Validação deve ser feita no frontend E backend

    "role": "VENDEDOR",

    "role": "VENDEDOR",
    "ativo": true,
    "createdAt": "2026-01-08T10:00:00+01:00",
    "updatedAt": null
  }
  
  Response (400 Bad Request):
  {
    "message": "Já existe um usuário com este telefone: 923987654",
    "timestamp": "2026-01-08T10:00:00+01:00"
  }

---

2.5. ATUALIZAR USUÁRIO
  Endpoint: PUT /api/v1/admin/usuarios/{id}
  
  Request Body:
  {
    "nome": "Maria Santos Silva",
    "telefone": "923987654",
    "senha": "novaSenha123",  // OPCIONAL (só enviar se quiser alterar)
    "role": "ADMIN",
    "ativo": true
  }
  
  Response (200 OK): Mesmo formato do GET por ID
  
  IMPORTANTE: Se não quiser alterar a senha, NÃO enviar o campo "senha"
              ou enviar como null/string vazia.

---

2.6. DESATIVAR USUÁRIO (Soft Delete)
  Endpoint: DELETE /api/v1/admin/usuarios/{id}
  
  Response (204 No Content): Sem corpo de resposta
  
  Observação: Usuário NÃO é removido do banco, apenas marcado como inativo.

---

2.7. REATIVAR USUÁRIO
  Endpoint: PATCH /api/v1/admin/usuarios/{id}/ativar
  
  Response (200 OK): Sem corpo de resposta

4.1. USUÁRIOS LIST VIEW (Tela Principal)
  
  Descrição: Página principal com listagem de todos os usuários
  
  Elementos:
    - Header com título "Gestão de Usuários"
    - Botão "Novo Usuário" (canto superior direito)
    - Barra de busca (nome/telefone)
    - Filtros:
      * Todos | Ativos | Inativos
      * Role: Todos | ADMIN | VENDEDOR | PORTEIRO | MONITORING
    - Tabela com colunas:
      * Nome
      * Telefone
      * Role (badge colorido)
      * Status (badge ativo/inativo)
      * Data de Criação
      * Ações (Ver, Editar, Desativar/Ativar)
    - Paginação (se houver muitos usuários)
    - Loading state
    - Empty state (quando não há usuários)

---

4.2. USUÁRIO FORM (Criar/Editar)
  
  Descrição: Formulário para criar ou editar usuário
  
  Campos:
    - Nome Completo (input text, obrigatório)
    - Telefone (input tel, obrigatório, único)
    - Senha (input password, obrigatório no create, opcional no edit)
    - Confirmar Senha (input password, deve corresponder à senha)
    - Role (select/dropdown, obrigatório)
      Opções: ADMIN, VENDEDOR, PORTEIRO, MONITORING
    - Status (switch/checkbox: Ativo/Inativo)
  
  Validações:
    - Nome: mínimo 3 caracteres
    - Telefone: formato válido (+244923456789 ou 923456789)
    - Senha: mínimo 6 caracteres
    - Confirmar senha: deve ser igual à senha
  
  Botões:
    - "Salvar" (submit)
    - "Cancelar" (voltar)

---

4.3. USUÁRIO TABLE
  
  Descrição: Tabela responsiva com lista de usuários
  
  Features:
    - Ordenação por colunas (nome, telefone, data)
    - Hover effects
    - Loading skeleton
    - Mensagem quando vazio
    - Ações inline (ícones/botões)

4.4. FILTROS DE BUSCA
  
  Descrição: Componente de filtros acima da tabela

OPÇÃO 1: PINIA STORE (Recomendado)

// stores/usuarioStore.js
import { defineStore } from 'pinia';
import usuarioService from '@/services/usuarioService';

export const useUsuarioStore = defineStore('usuario', {
  state: () => ({
    usuarios: [],
    currentUsuario: null,
    loading: false,
    error: null,
    filters: {
      search: '',
      status: 'todos', // 'todos' | 'ativos' | 'inativos'
      role: null
    }
  }),
  
  getters: {
    usuariosFiltrados: (state) => {
      let filtered = state.usuarios;
      
      // Filtrar por busca
      if (state.filters.search) {
        const search = state.filters.search.toLowerCase();
        filtered = filtered.filter(u => 
          u.nome.toLowerCase().includes(search) ||
          u.telefone.includes(search)
        );
      }
      
      // Filtrar por status
      if (state.filters.status === 'ativos') {
        filtered = filtered.filter(u => u.ativo === true);
      } else if (state.filters.status === 'inativos') {
        filtered = filtered.filter(u => u.ativo === false);
      }
      
      // Filtrar por role
      if (state.filters.role) {
        filtered = filtered.filter(u => u.role === state.filters.role);
      }
      
      return filtered;
    },
    
    usuariosAtivos: (state) => state.usuarios.filter(u => u.ativo),
    usuariosInativos: (state) => state.usuarios.filter(u => !u.ativo),
    totalUsuarios: (state) => state.usuarios.length
  },
  
  actions: {
    async fetchUsuarios() {
      this.loading = true;
      this.error = null;
      try {
        this.usuarios = await usuarioService.getAll();
      } catch (error) {
        this.error = error.message;
        throw error;
      } finally {
        this.loading = false;
      }
    },
    
    async fetchUsuarioById(id) {
      this.loading = true;
      this.error = null;
      try {
        this.currentUsuario = await usuarioService.getById(id);
        return this.currentUsuario;
      } catch (error) {
        this.error = error.message;
        throw error;
      } finally {
        this.loading = false;
      }
    },
    
    async createUsuario(data) {
      this.loading = true;
      this.error = null;
      try {
        const newUsuario = await usuarioService.create(data);
        this.usuarios.push(newUsuario);
        return newUsuario;
      } catch (error) {
        this.error = error.message;
        throw error;
      } finally {
        this.loading = false;
      }
    },
    
    async updateUsuario(id, data) {
      this.loading = true;
      this.error = null;
      try {
        const updatedUsuario = await usuarioService.update(id, data);
        const index = this.usuarios.findIndex(u => u.id === id);
        if (index !== -1) {
          this.usuarios[index] = updatedUsuario;
        }
        return updatedUsuario;
      } catch (error) {
        this.error = error.message;
        throw error;
      } finally {
        this.loading = false;
      }
    },
    
    async deactivateUsuario(id) {
      this.loading = true;
      this.error = null;
      try {
        await usuarioService.deactivate(id);
        const usuario = this.usuarios.find(u => u.id === id);
        if (usuario) {
          usuario.ativo = false;
        }
      } catch (error) {
        this.error = error.message;
        throw error;
      } finally {
        this.loading = false;
      }
    },
    
    async activateUsuario(id) {
      this.loading = true;
      this.error = null;
      try {
        await usuarioService.activate(id);
        const usuario = this.usuarios.find(u => u.id === id);
        if (usuario) {
          usuario.ativo = true;
        }
      } catch (error) {
        this.error = error.message;
        throw error;
      } finally {
        this.loading = false;
      }
    },
    
    setFilter(filterType, value) {
      this.filters[filterType] = value;
    },
    
    clearFilters() {
      this.filters = {
        search: '',
        status: 'todos',
        role: null
      };
    }
  }
});

// utils/validators/usuarioValidators.js

export const validarNome = (nome) => {
  if (!nome || nome.trim().length < 3) {
    return 'O nome deve ter pelo menos 3 caracteres';
  }
  return null;
};

export const validarTelefone = (telefone) => {
  if (!telefone) {
    return 'O telefone é obrigatório';
  }
  
  // Remover espaços e caracteres especiais
  const cleaned = telefone.replace(/\D/g, '');
  
  // Validar formato angolano (9 dígitos)
  if (cleaned.length === 9 && cleaned.startsWith('9')) {
    return null;
  }
  
  // Validar com código do país (+244)
  if (cleaned.length === 12 && cleaned.startsWith('244')) {
    return null;
  }
  
  return 'Telefone inválido. Use formato: 923456789 ou +244923456789';
};

export const validarSenha = (senha, isEdit = false) => {
  // Senha é opcional na edição
  if (isEdit && (!senha || senha.trim() === '')) {
    return null;
  }
  
  if (!senha || senha.length < 6) {
    return 'A senha deve ter pelo menos 6 caracteres';
  }
  
  return null;
};

export const validarConfirmacaoSenha = (senha, confirmacao) => {
  if (senha !== confirmacao) {
    return 'As senhas não correspondem';
  }
  return null;
};

export const validarRole = (role) => {
  const rolesValidos = ['ADMIN', 'VENDEDOR', 'PORTEIRO', 'MONITORING'];
  
  if (!role || !rolesValidos.includes(role)) {
    return 'Selecione um perfil válido';
  }
  
  return null;
};

USAR NO FORMULÁRIO:

const errors = ref({
  nome: null,
  telefone: null,

const validateForm = () => {

};

9. FLUXOS DE USUÁRIO
================================================================================

FLUXO 1: CRIAR NOVO USUÁRIO

1. Admin clica em "Novo Usuário"
2. Sistema abre formulário em branco
3. Admin preenche:
   - Nome: "João Silva"
   - Telefone: "923456789"
   - Senha: "senha123"
   - Confirmar Senha: "senha123"
   - Role: "VENDEDOR"
   - Status: Ativo (marcado)
4. Admin clica em "Salvar"
5. Sistema valida:
   - Todos os campos obrigatórios preenchidos? ✓
   - Senhas correspondem? ✓
   - Telefone válido? ✓
6. Sistema envia POST /api/v1/admin/usuarios
7. Backend responde:
   - Sucesso (201): Sistema mostra toast "Usuário criado com sucesso"
   - Erro (400): Sistema mostra mensagem de erro específica
8. Sistema redireciona para lista de usuários
9. Novo usuário aparece na tabela

---

FLUXO 2: EDITAR USUÁRIO EXISTENTE

1. Admin visualiza lista de usuários
2. Admin clica em ícone "Editar" na linha do usuário
3. Sistema carrega dados do usuário
4. Sistema pré-preenche formulário:
   - Nome: "João Silva"
   - Telefone: "923456789"
   - Senha: (campo vazio - não mostrar senha atual)
   - Role: "VENDEDOR"
   - Status: Ativo
5. Admin altera apenas o que deseja (ex: nome para "João Silva Santos")
6. Admin clica em "Salvar"
7. Sistema valida mudanças
8. Sistema envia PUT /api/v1/admin/usuarios/{id}
   Importante: Se senha estiver vazia, não enviar campo "senha"
9. Backend responde:
   - Sucesso (200): Toast "Usuário atualizado com sucesso"
   - Erro (400/404): Mensagem de erro específica
10. Sistema atualiza lista
11. Usuário vê dados atualizados na tabela

---

FLUXO 3: DESATIVAR USUÁRIO

1. Admin visualiza lista de usuários
2. Admin clica em ícone "Desativar" na linha do usuário ativo
3. Sistema abre modal de confirmação:
   "Tem certeza que deseja desativar o usuário João Silva?"
4. Admin clica em "Confirmar"
5. Sistema envia DELETE /api/v1/admin/usuarios/{id}
6. Backend responde:
   - Sucesso (204): Toast "Usuário desativado"
   - Erro: Mensagem de erro
7. Sistema atualiza status na lista
8. Badge do usuário muda para "Inativo" (cinza)
9. Botão muda para "Ativar"

---

FLUXO 4: REATIVAR USUÁRIO

1. Admin filtra por "Inativos" ou vê usuário inativo na lista
2. Admin clica em "Ativar"
3. Sistema abre modal de confirmação:
   "Tem certeza que deseja reativar o usuário João Silva?"
4. Admin confirma
5. Sistema envia PATCH /api/v1/admin/usuarios/{id}/ativar
6. Backend responde sucesso
7. Toast "Usuário reativado"
8. Status atualiza para "Ativo"

---

FLUXO 5: BUSCAR/FILTRAR USUÁRIOS

1. Admin acessa lista de usuários (mostra todos)
2. Admin digita "João" no campo de busca
3. Sistema filtra automaticamente (debounce 300ms)
4. Tabela mostra apenas usuários com "João" no nome/telefone
5. Admin clica em filtro "Role: VENDEDOR"
6. Sistema aplica filtro adicional
7. Tabela mostra apenas vendedores com "João" no nome
8. Admin clica em "Limpar Filtros"
9. Sistema volta a mostrar todos os usuários

TIPOS DE NOTIFICAÇÃO (Toast):

TRATAMENTO DE ERROS NO SERVICE:

try {
  const response = await api.post('/api/v1/admin/usuarios', data);
  return response.data;
} catch (error) {
  if (error.response) {
    // Erro do backend
    throw new Error(error.response.data.message || 'Erro ao criar usuário');
  } else if (error.request) {
    // Sem resposta do servidor
    throw new Error('Sem resposta do servidor. Verifique sua conexão.');
  } else {
    // Erro ao configurar requisição
    throw new Error('Erro ao processar requisição');
  }
}

EXEMPLO 1: SERVICE (API)

// services/usuarioService.js
import api from '@/utils/api'; // Axios instance com interceptors

const BASE_URL = '/api/v1/admin/usuarios';

export default {
  async getAll() {
    const response = await api.get(BASE_URL);
    return response.data;
  },
  
  async getActives() {
    const response = await api.get(`${BASE_URL}/ativos`);
    return response.data;
  },
  
  async getById(id) {
    const response = await api.get(`${BASE_URL}/${id}`);
    return response.data;
  },
  
  async create(data) {
    const response = await api.post(BASE_URL, data);
    return response.data;
  },
  
  async update(id, data) {
    // Remover senha se vazia
    const payload = { ...data };
    if (!payload.senha || payload.senha.trim() === '') {
      delete payload.senha;
    }
    
    const response = await api.put(`${BASE_URL}/${id}`, payload);
    return response.data;
  },
  
  async deactivate(id) {
    await api.delete(`${BASE_URL}/${id}`);
  },
  
  async activate(id) {
    await api.patch(`${BASE_URL}/${id}/ativar`);
  }
};

EXEMPLO 2: COMPONENTE DE LISTAGEM

<template>
  <div class="usuarios-list">
    <!-- Header -->
    <div class="header">
      <h1>Gestão de Usuários</h1>
      <button @click="goToCreate" class="btn-primary">
        + Novo Usuário
      </button>
    </div>
    
    <!-- Filtros -->
    <UsuarioFilters
      v-model:search="filters.search"
      v-model:status="filters.status"
      v-model:role="filters.role"
      @clear="clearFilters"
    />
    
    <!-- Loading -->
    <div v-if="loading" class="loading">
      <Spinner />
      <p>Carregando usuários...</p>
    </div>
    
    <!-- Tabela -->
    <UsuarioTable
      v-else
      :usuarios="usuariosFiltrados"
      @edit="goToEdit"
      @view="goToView"
      @deactivate="confirmDeactivate"
      @activate="confirmActivate"
    />
    
    <!-- Empty State -->
    <div v-if="!loading && usuariosFiltrados.length === 0" class="empty">
      <p>Nenhum usuário encontrado</p>
    </div>
    
    <!-- Modal de Confirmação -->
    <ConfirmModal
      v-model="showConfirmModal"
      :title="confirmTitle"
      :message="confirmMessage"
      @confirm="handleConfirm"
    />
  </div>
</template>

<script setup>
import { ref, computed, onMounted } from 'vue';
import { useRouter } from 'vue-router';
import { useUsuarioStore } from '@/stores/usuarioStore';
import { showToast } from '@/utils/toast';
import UsuarioFilters from '@/components/admin/usuarios/UsuarioFilters.vue';
import UsuarioTable from '@/components/admin/usuarios/UsuarioTable.vue';
import ConfirmModal from '@/components/common/ConfirmModal.vue';

const router = useRouter();
const usuarioStore = useUsuarioStore();

const loading = computed(() => usuarioStore.loading);
const usuariosFiltrados = computed(() => usuarioStore.usuariosFiltrados);

const filters = ref({
  search: '',
  status: 'todos',
  role: null
});

const showConfirmModal = ref(false);
const confirmAction = ref(null);
const confirmUserId = ref(null);
const confirmTitle = ref('');
const confirmMessage = ref('');

onMounted(async () => {
  try {
    await usuarioStore.fetchUsuarios();
  } catch (error) {
    showToast('Erro ao carregar usuários', 'error');
  }
});

const goToCreate = () => {
  router.push({ name: 'UsuarioCreate' });
};

const goToEdit = (usuario) => {
  router.push({ name: 'UsuarioEdit', params: { id: usuario.id } });
};

const goToView = (usuario) => {
  router.push({ name: 'UsuarioDetail', params: { id: usuario.id } });
};

const confirmDeactivate = (usuario) => {
  confirmAction.value = 'deactivate';
  confirmUserId.value = usuario.id;
  confirmTitle.value = 'Desativar Usuário';
  confirmMessage.value = `Tem certeza que deseja desativar o usuário ${usuario.nome}?`;
  showConfirmModal.value = true;
};

const confirmActivate = (usuario) => {
  confirmAction.value = 'activate';
  confirmUserId.value = usuario.id;
  confirmTitle.value = 'Reativar Usuário';
  confirmMessage.value = `Tem certeza que deseja reativar o usuário ${usuario.nome}?`;
  showConfirmModal.value = true;
};

const handleConfirm = async () => {
  try {
    if (confirmAction.value === 'deactivate') {
      await usuarioStore.deactivateUsuario(confirmUserId.value);
      showToast('Usuário desativado com sucesso', 'success');
    } else if (confirmAction.value === 'activate') {
      await usuarioStore.activateUsuario(confirmUserId.value);
      showToast('Usuário reativado com sucesso', 'success');
    }
  } catch (error) {
    showToast(error.message, 'error');
  }
  
  showConfirmModal.value = false;
};

const clearFilters = () => {
  filters.value = {
    search: '',
    status: 'todos',
    role: null
  };
  usuarioStore.clearFilters();
};
</script>

<template>
  <div class="usuario-form">
    <div class="header">
      <button @click="goBack" class="btn-back">← Voltar</button>
      <h1>{{ isEdit ? 'Editar Usuário' : 'Novo Usuário' }}</h1>

    <form @submit.prevent="submitForm">

          v-model="form.confirmacaoSenha"
          type="password"
          placeholder="Confirme a senha"
          :class="{ 'error': errors.confirmacaoSenha }"
        />
        <span v-if="errors.confirmacaoSenha" class="error-message">
          {{ errors.confirmacaoSenha }}
        </span>
      </div>
      
      <!-- Role -->
      <div class="form-group">
        <label for="role">Perfil de Acesso *</label>
        <select
          id="role"
          v-model="form.role"
          :class="{ 'error': errors.role }"
        >
          <option value="">Selecione...</option>
          <option value="ADMIN">Administrador (acesso total)</option>
          <option value="VENDEDOR">Vendedor (vendas presenciais)</option>
          <option value="PORTEIRO">Porteiro (validação de bilhetes)</option>

          {{ loading ? 'Salvando...' : 'Salvar Usuário' }}
        </button>
      </div>
    </form>
  </div>

<script setup>
import { ref, onMounted, computed } from 'vue';
import { useRouter, useRoute } from 'vue-router';
import { useUsuarioStore } from '@/stores/usuarioStore';
import { showToast } from '@/utils/toast';
import {
  validarNome,
  validarTelefone,
  validarSenha,
  validarConfirmacaoSenha,
  validarRole
} from '@/utils/validators/usuarioValidators';

const router = useRouter();
const route = useRoute();
const usuarioStore = useUsuarioStore();

const isEdit = computed(() => !!route.params.id);
const loading = ref(false);

const form = ref({
  nome: '',
  telefone: '',
  senha: '',
  confirmacaoSenha: '',
  role: '',
  ativo: true
});

const errors = ref({
  nome: null,
  telefone: null,
  senha: null,
  confirmacaoSenha: null,
  role: null
});

onMounted(async () => {
  if (isEdit.value) {
    try {
      const usuario = await usuarioStore.fetchUsuarioById(route.params.id);
      form.value = {
        nome: usuario.nome,
        telefone: usuario.telefone,
        senha: '',
        confirmacaoSenha: '',
        role: usuario.role,
        ativo: usuario.ativo
      };
    } catch (error) {
      showToast('Erro ao carregar usuário', 'error');
      goBack();
    }
  }
});

const validateForm = () => {
  errors.value.nome = validarNome(form.value.nome);
  errors.value.telefone = validarTelefone(form.value.telefone);
  errors.value.senha = validarSenha(form.value.senha, isEdit.value);
  
  if (form.value.senha) {
    errors.value.confirmacaoSenha = validarConfirmacaoSenha(
      form.value.senha,
      form.value.confirmacaoSenha
    );
  }
  
  errors.value.role = validarRole(form.value.role);
  
  return !Object.values(errors.value).some(error => error !== null);
};

const submitForm = async () => {
  if (!validateForm()) {
    showToast('Por favor, corrija os erros no formulário', 'error');
    return;
  }
  
  loading.value = true;
  
  try {
    const payload = {
      nome: form.value.nome,
      telefone: form.value.telefone,
      senha: form.value.senha || undefined, // Não enviar se vazio
      role: form.value.role,
      ativo: form.value.ativo
    };
    
    if (isEdit.value) {
      await usuarioStore.updateUsuario(route.params.id, payload);
      showToast('Usuário atualizado com sucesso!', 'success');
    } else {
      await usuarioStore.createUsuario(payload);
      showToast('Usuário criado com sucesso!', 'success');
    }
    
    router.push({ name: 'Usuarios' });
  } catch (error) {
    showToast(error.message, 'error');
  } finally {
    loading.value = false;
  }
};

const goBack = () => {
  router.push({ name: 'Usuarios' });
};
</script>
